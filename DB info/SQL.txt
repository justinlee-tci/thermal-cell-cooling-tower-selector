-- Keep-alive heartbeat. Supabase pauses a free-tier project after ~7 days of
-- inactivity; scripts/keep-alive.mjs writes a row here every 3 days (driven by
-- .github/workflows/keep-alive.yml) so the project never goes idle that long.
-- Written only with the service_role key, so no RLS policy is needed.
create table if not exists public.keep_alive (
  id bigint generated by default as identity primary key,
  pinged_at timestamp with time zone
) TABLESPACE pg_default;
-- The live table allows NULL and has no default (row id 42 has a null
-- pinged_at from a bare insert). The script always sends an explicit
-- timestamp, so this is harmless; adding `default now()` would prevent it:
--   alter table public.keep_alive alter column pinged_at set default now();

-- ===== ENABLE RLS ON ALL TABLES =====
